def render_modalities(...):

    current_dir = str(source_scene.parent.resolve())
    if (_last_source_scene_dir
            and _last_source_scene_dir != current_dir):

            _RESIDENT_SCENE_CACHE.clear()
    _last_source_scene_dir = current_dir
    ...

설계 판단 포인트

왜 worker 전체 캐시를 지우나?
같은 worker 프로세스가 glass_corridor → office_lobby로 전환하는 경우에만 _last_source_scene_dir가 달라진다. 다른 worker 프로세스는 각자 별도의 모듈 인스턴스를 갖기 때문에 독립적이며 올바른 캐시를 유지한다.
캐시 히트는 버그가 아니다
같은 worker가 office_lobby를 두 번 연달아 렌더하면 두 번째 요청은 정당하게 캐시 히트한다. 씬 자체는 변하지 않았기 때문. 문제는 다른 씬으로 전환했을 때 캐시가 남는 것.

Emitter 감지

def _has_emitter(root: ET.Element) -> bool:
    # 최상위 emitter 태그
    if root.find("./emitter") is not None:
        return True
    # shape 내부 area light
    return any(
        shape.find("./emitter") is not None
        for shape in root.findall("./shape")
    )

상수 조명 주입

def _inject_constant_emitter(
        root: ET.Element, radiance: float
) -> None:
    emitter = ET.SubElement(root, "emitter",
                            {"type": "constant"})
    ET.SubElement(emitter, "rgb", {
        "name": "radiance",
        "value": f"{float(radiance):.6f}"
    })
# 결과: 무한 구면에서 균일한 환경광 방사

export const deleteOpticalNavObservations = (
    projectId: string,
    sceneId: string,
    nodeIds: string[] | null
) =>
    fetch(`${opticalProject(projectId)}/scenes/`
        + `${encodeURIComponent(sceneId)}/observations`,
    {
        method: 'DELETE',
        headers: nodeIds
            ? { 'Content-Type': 'application/json' } : {},
        body: nodeIds
            ? JSON.stringify({ node_ids: nodeIds })
            : undefined
    }).then(json);

infer_material_kind()

# robomituba_bridge/material_mapping.py
def infer_material_kind(material_id: str, extras: dict)
    -> str:
  # 1. extras["material_kind"] 명시값 우선
  # 2. material_id 접두사 패턴 매칭:
  #    glass_* / crystal_* → "glass"
  #    metal_* / steel_* / copper_* → "metal"
  #    plastic_* → "plastic"
  #    wood_* / fabric_* → "diffuse"
  # 3. roughness > 0.9 → "diffuse"
  # 4. metallic > 0.5 → "metal"
  # 5. transmission > 0.5 → "glass"
  return kind  # "diffuse" | "plastic" | "metal" | "glass"

hpBRDF / pBRDF → Mitsuba BSDF 변환
KAIST SIGGRAPH Asia 2025 측정 편광 BRDF · .pbrdf 채널 파일 · measured_polarized BSDF
hpBRDF (Hyperspectral Polarized BRDF): KAIST RGL이 SIGGRAPH Asia 2025에 발표한 실측 편광 BRDF 데이터셋. 파장별(RGB+NIR) 스토크스 파라미터를 실제 측정한 값으로 제공. Mitsuba 3의 measured_polarized BSDF 타입과 연동 (패치 필요).

데이터 구조 (.pbrdf 채널 파일)

data/hpbrdf_2025/channels/{material_id}/
  ├── r.pbrdf       # Red 채널 (780nm 근방)
  ├── g.pbrdf       # Green 채널 (550nm 근방)
  ├── b.pbrdf       # Blue 채널 (470nm 근방)
  └── nir.pbrdf     # NIR 채널 (850nm 근방, 선택)

# 각 .pbrdf: 이진 blob
#   - 반구 샘플 그리드 (theta_i × phi_i × theta_o × phi_o)
#   - 각 샘플: 4×4 Mueller matrix (float32)
#   - 전체 크기: ~수십 MB / 채널

# Variant 요구: mono_polarized / scalar_mono_polarized
# (rgb variant는 단일 채널만 — polarization 미지원)

BSDF 샘플링 흐름 (내부)

eval(wi, wo, si):
  # 1. (θi, φi) 로 BRDF 그리드 인덱스 계산
  #    θi = acos(dot(wi, n)), φi = atan2(...)
  # 2. r/g/b 채널 각각 Mueller 보간
  #    M_r = bilinear(grid_r, θi, φi, θo, φo)
  # 3. 입사 스토크스 벡터 Si 에 M_r 적용
  #    So_r = M_r @ Si
  # 4. Stokes 채널별 분리 → RGB 최종 radiance
  # → mi.Spectrum 반환 (polarized 빌드에서만 유효)

실험 갈림길 — Route A vs Route B

Route A — End-to-end VLN/VLA
입력: RGB observation + instruction + current node
출력: next action / next node / stop
보고 싶은 것: RGB-only agent가 유리벽을 free space로 잘못 판단하나?
데이터 필요: rgb 렌더 · episode JSON
Route B — Detection + Planner
입력: RGB / NIR / hazard_mask prediction
중간 출력: transparent obstacle map
planner: A* / Dijkstra
보고 싶은 것: hazard detection을 붙이면 collision이 줄어드나?
데이터 필요: hazard_mask oracle · graph edges
미팅 결론: Route B (detection + 기존 planner)가 우리 optical cue의 downstream 기여를 더 직접적으로 증명할 수 있다. End-to-end에만 집착하지 말고 Route B도 feasibility test 대상으로 열어두기.
